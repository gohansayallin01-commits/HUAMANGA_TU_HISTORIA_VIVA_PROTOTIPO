import React, { useState } from 'react';
import { Link, useNavigate } from 'react-router-dom';
import { useI18n } from '../../i18n/i18nContext';
import { storageService } from '../../services/storageService';
import { useToast } from './ToastContext';
import { 
  Bell, 
  Globe, 
  Sparkles, 
  RotateCcw, 
  Briefcase, 
  ShieldCheck, 
  Menu, 
  X, 
  Check, 
  Compass, 
  AlertTriangle,
} from 'lucide-react';

interface HeaderProps {
  onStartTour?: () => void;
}

export const Header: React.FC<HeaderProps> = ({ onStartTour }) => {
  const { language, setLanguage, t } = useI18n();
  const { showToast } = useToast();
  const navigate = useNavigate();

  const [showLangMenu, setShowLangMenu] = useState(false);
  const [showNotifications, setShowNotifications] = useState(false);
  const [showNavDrawer, setShowNavDrawer] = useState(false);

  const notifications = storageService.getNotifications();
  const unreadCount = notifications.filter(n => !n.read).length;

  // Saludo según la hora del día
  const getGreeting = () => {
    const hour = new Date().getHours();
    if (hour < 12) return t('greetings.morning');
    if (hour < 19) return t('greetings.afternoon');
    return t('greetings.evening');
  };

  const handleLanguageSelect = (lang: 'es' | 'en' | 'fr' | 'pt') => {
    if (lang === 'fr' || lang === 'pt') {
      showToast(`${lang.toUpperCase()} estará disponible próximamente en la versión productiva.`, 'info');
      return;
    }
    setLanguage(lang);
    setShowLangMenu(false);
    showToast(lang === 'es' ? 'Idioma cambiado a Español' : 'Language changed to English', 'success');
  };

  const handleExhibitionMode = () => {
    storageService.resetAllToExhibitionState();
  };

  Bell, 
  Globe, 
  Sparkles, 
  RotateCcw, 
  Briefcase, 
  ShieldCheck, 
  Menu, 
  X, 
  Check, 
  Compass, 
  AlertTriangle,
} from 'lucide-react';
